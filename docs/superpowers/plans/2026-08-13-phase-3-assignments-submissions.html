namespace TeachHub.Domain.Assignments;

public class Rubric : Entity
{
    public required string Name { get; set; }
    public required Guid InstructorId { get; set; }      // Gap #18: rubric tái sử dụng giữa các môn
    public bool IsTemplate { get; set; }
    public List<RubricCriterion> Criteria { get; set; } = [];

    public decimal MaxPoints => Criteria.Sum(c => c.MaxPoints);

    /// <summary>
    /// scores: criterion_id → điểm. Tiêu chí thiếu được tính 0 (chấm dở dang là hợp lệ).
    /// Toàn bộ tính toán dùng decimal — điểm số không bao giờ được đi qua double.
    /// </summary>
    public decimal Score(IReadOnlyDictionary<Guid, decimal> scores)
    {
        decimal tong = 0;
        foreach (var c in Criteria)
        {
            if (!scores.TryGetValue(c.Id, out var d)) continue;

            if (d < 0 || d > c.MaxPoints)
                throw new ArgumentOutOfRangeException(nameof(scores),
                    $"Điểm tiêu chí '{c.Description}' phải nằm trong [0, {c.MaxPoints}], nhận được {d}.");

            tong += d;
        }
        return Math.Round(tong, 2, MidpointRounding.AwayFromZero);
    }
}

namespace TeachHub.Domain.Assignments;

public enum SlotOwnerType { Assignment, FinalAssessment }

public class SubmissionSlot : Entity
{
    public SlotOwnerType OwnerType { get; set; } = SlotOwnerType.Assignment;

    // ADR-003: hai FK thật thay cho owner_id đa hình, có CHECK constraint ở DB.
    public Guid? AssignmentId { get; set; }
    public Guid? FinalAssessmentId { get; set; }

    public int OrderIndex { get; set; }
    public required string SlotKey { get; set; }        // tự đặt: source_code, images, report...
    public required string Label { get; set; }
    public List<string> AcceptedFileTypes { get; set; } = [];
    public bool AllowLinkInsteadOfFile { get; set; }
    public int MaxFiles { get; set; } = 1;
    public int MaxFileSizeMb { get; set; } = 50;
    public bool IsRequired { get; set; }
    public RequiredStructure? RequiredStructure { get; set; }   // jsonb, chỉ với slot .zip

    public bool Accepts(string fileName)
    {
        var ext = Path.GetExtension(fileName);
        return AcceptedFileTypes.Any(t => string.Equals(t, ext, StringComparison.OrdinalIgnoreCase));
    }

    public bool HasRoomFor(int currentCount) => currentCount < MaxFiles;

    public bool AcceptsZip =>
        AcceptedFileTypes.Any(t => string.Equals(t, ".zip", StringComparison.OrdinalIgnoreCase));

    public SubmissionSlot CloneForNewOwner() => new()
    {
        OwnerType = OwnerType, OrderIndex = OrderIndex,
        SlotKey = SlotKey, Label = Label,
        AcceptedFileTypes = [.. AcceptedFileTypes],
        AllowLinkInsteadOfFile = AllowLinkInsteadOfFile,
        MaxFiles = MaxFiles, MaxFileSizeMb = MaxFileSizeMb, IsRequired = IsRequired,
        RequiredStructure = RequiredStructure?.DeepCopy(),
    };
}

    private static MemoryStream TaoZip(params (string Path, byte[] Content)[] entries)
    {
        var ms = new MemoryStream();
        using (var zip = new ZipArchive(ms, ZipArchiveMode.Create, leaveOpen: true))
            foreach (var (path, content) in entries)
            {
                using var s = zip.CreateEntry(path).Open();
                s.Write(content);
            }
        ms.Position = 0;
        return ms;

public sealed class UnsafeArchiveException(string message) : Exception(message);

public sealed record ZipLimits
{
    public long MaxTotalUncompressedBytes { get; init; } = 200L * 1024 * 1024;   // 200 MB
    public int MaxEntryCount { get; init; } = 5_000;
    public int MaxCompressionRatio { get; init; } = 100;
    public int MaxPathDepth { get; init; } = 20;

    public static ZipLimits Default => new();
}

/// <summary>
/// Đọc cấu trúc file zip mà KHÔNG BAO GIỜ ghi ra đĩa theo đường dẫn nằm trong file zip.
/// Mọi entry đều bị kiểm tra trước khi đọc; vượt hạn mức là dừng ngay, không đọc tiếp.
/// Xem ADR-005.
/// </summary>
public static class SafeZipReader
{
    private static readonly string[] RacHeThong = ["__MACOSX/", ".DS_Store", "Thumbs.db", "desktop.ini"];

    public static FileTree ReadTree(Stream zipStream, ZipLimits limits)
    {
        ZipArchive archive;
        try
        {
            archive = new ZipArchive(zipStream, ZipArchiveMode.Read, leaveOpen: true);
        }
        catch (InvalidDataException e)
        {
            throw new UnsafeArchiveException($"File không phải định dạng ZIP hợp lệ: {e.Message}");
        }

        using (archive)
        {
            if (archive.Entries.Count > limits.MaxEntryCount)
                throw new UnsafeArchiveException(
                    $"File nén có {archive.Entries.Count} mục, vượt giới hạn {limits.MaxEntryCount}. " +
                    "Có thể bạn đã nén nhầm cả thư mục node_modules hoặc bin/obj.");

            long tongGiaiNen = 0, tongNen = 0;
            var files = new List<FileTreeNode>();

            foreach (var entry in archive.Entries)
            {
                if (entry.FullName.EndsWith('/')) continue;                       // thư mục
                if (RacHeThong.Any(r => entry.FullName.Contains(r, StringComparison.Ordinal))) continue;

                var duongDan = NormalizeAndGuard(entry.FullName, limits);

                tongGiaiNen += entry.Length;
                tongNen     += entry.CompressedLength;

                if (tongGiaiNen > limits.MaxTotalUncompressedBytes)
                    throw new UnsafeArchiveException(
                        $"Nội dung sau giải nén vượt {limits.MaxTotalUncompressedBytes / 1024 / 1024} MB.");

                files.Add(new FileTreeNode(duongDan, entry.Length));
            }

            // Kiểm tỉ lệ nén sau cùng: từng entry lẻ có thể bình thường
            // nhưng tổng thể vẫn là một quả bom.
            if (tongNen > 0 && tongGiaiNen / tongNen > limits.MaxCompressionRatio)
                throw new UnsafeArchiveException(
                    $"Tỉ lệ nén {tongGiaiNen / tongNen}:1 bất thường, từ chối xử lý.");

            return new FileTree(files);
        }
    }

    private static string NormalizeAndGuard(string raw, ZipLimits limits)
    {
        var path = raw.Replace('\\', '/');

        if (Path.IsPathRooted(path) || path.StartsWith('/'))
            throw new UnsafeArchiveException($"Đường dẫn tuyệt đối trong file nén: '{raw}'.");

        var parts = path.Split('/', StringSplitOptions.RemoveEmptyEntries);

        if (parts.Contains(".."))
            throw new UnsafeArchiveException($"Đường dẫn thoát khỏi thư mục đích (zip-slip): '{raw}'.");

        if (parts.Length > limits.MaxPathDepth)
            throw new UnsafeArchiveException($"Đường dẫn quá sâu ({parts.Length} cấp): '{raw}'.");

        return string.Join('/', parts);
    }
}

public sealed record FileTreeNode(string Path, long SizeBytes);

public sealed record StructureCheckResult(bool Passed, List<string> Missing);

public sealed class FileTree(List<FileTreeNode> files)
{
    public List<FileTreeNode> Files { get; } = files;

    public StructureCheckResult CheckStructure(RequiredStructure required)
    {
        // Nếu MỌI file đều nằm dưới cùng một thư mục gốc, coi thư mục đó là gốc thật.
        // Sinh viên nén bằng cách chuột phải vào thư mục là hành vi mặc định của Windows
        // và macOS — chặn nó sẽ gắn cờ đỏ oan cho nửa lớp.
        var goc = CommonRootFolder();
        var duongDan = goc is null
            ? Files.Select(f => f.Path).ToList()
            : Files.Select(f => f.Path[(goc.Length + 1)..]).ToList();

        var thieu = new List<string>();

        foreach (var f in required.RequiredFiles)
            if (!duongDan.Any(p => string.Equals(p, f, StringComparison.OrdinalIgnoreCase)))
                thieu.Add(f);

        foreach (var d in required.RequiredFolders)
            if (!duongDan.Any(p => p.StartsWith(d + "/", StringComparison.OrdinalIgnoreCase)))
                thieu.Add(d + "/");

        return new StructureCheckResult(thieu.Count == 0, thieu);
    }

    private string? CommonRootFolder()
    {
        if (Files.Count == 0) return null;

        var dau = Files[0].Path.Split('/');
        if (dau.Length < 2) return null;

        var ungVien = dau[0];
        return Files.All(f => f.Path.StartsWith(ungVien + "/", StringComparison.Ordinal))
            ? ungVien : null;
    }

namespace TeachHub.Infrastructure.Jobs;

public sealed class ProcessSubmissionFileJob(
    AppDbContext db, IFileStorage storage, ILogger<ProcessSubmissionFileJob> log)
{
    /// <summary>
    /// Chạy trên queue "zip" với ít worker (ADR-005): một file nén độc hại
    /// không được phép chiếm hết worker và làm treo mọi job khác.
    /// </summary>
    [Queue(JobsDependencyInjection.ZipQueue)]
    [AutomaticRetry(Attempts = 2)]
    public async Task RunAsync(Guid submissionFileId, CancellationToken ct)
    {
        var file = await db.SubmissionFiles
            .Include(f => f.Slot)
            .SingleOrDefaultAsync(f => f.Id == submissionFileId, ct);

        if (file is null) return;                        // đã bị xóa trong lúc chờ
        if (!file.Slot.AcceptsZip) { file.ProcessingStatus = FileProcessingStatus.Done; }
        else
        {
            try
            {
                await using var stream = await storage.OpenReadAsync(file.FilePath, ct);
                var tree = SafeZipReader.ReadTree(stream, ZipLimits.Default);

                file.FileTree = tree;
                file.StructureCheckPassed = file.Slot.RequiredStructure is null
                    ? null
                    : tree.CheckStructure(file.Slot.RequiredStructure).Passed;
                file.ProcessingStatus = FileProcessingStatus.Done;
            }
            catch (UnsafeArchiveException e)
            {
                // Lỗi mong đợi: file hỏng hoặc độc hại. Ghi nhận rồi đi tiếp,
                // KHÔNG ném lại — ném lại sẽ khiến Hangfire retry một file
                // vốn dĩ không bao giờ xử lý được.
                log.LogWarning("File nộp {Id} bị từ chối: {Reason}", file.Id, e.Message);
                file.ProcessingStatus = FileProcessingStatus.Failed;
                file.ProcessingError = e.Message;
            }
        }

        await db.SaveChangesAsync(ct);
        await UpdateSubmissionStatusAsync(file.SubmissionId, ct);
    }
}

    Response.Headers.CacheControl = "no-cache";

    while (!ct.IsCancellationRequested)
    {

        await Response.WriteAsync($"data: {JsonSerializer.Serialize(tt)}\n\n", ct);
