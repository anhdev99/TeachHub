Kiến trúc API — Controllers + MediatR

Phase này không có endpoint kiểu Minimal API nào. Mọi đường vào hệ thống là một action trong controller gắn [ApiController], kế thừa ApiControllerBase. Controller làm đúng ba việc: định tuyến, lấy CurrentUserId từ claim, gọi sender.Send rồi quy đổi Result. Không có if nghiệp vụ, không truy vấn DbContext, không try/catch trong controller — toàn bộ nghiệp vụ nằm ở Command/Query handler của tầng Application.

Hợp đồng API ở SRS mục 7 không đổi một ký tự nào: cùng đường dẫn, cùng payload, cùng mã lỗi. Chỉ đổi cách hiện thực.

Hai lớp chặn dữ liệu sai, và chúng không thay nhau được:
• Sai hình dạng đầu vào (điểm âm, nhận xét dài quá, group_size nhỏ hơn 2) → FluentValidation viết trên chính command. Nó chạy trong ValidationBehavior, tức là trước TransactionBehavior: một request sai dữ liệu không bao giờ mở transaction rồi rollback vô ích.
• Sai theo dữ liệu đang có (điểm vượt trần của tiêu chí rubric, sinh viên không thuộc nhóm nào, bài đã có bản nộp) → handler đọc DB rồi trả Result, vì đó là một phần hợp đồng của use case và phải hiện ra trong chữ ký hàm chứ không giấu vào exception.
[Authorize] trên class và trên action CỘNG DỒN, không ghi đè nhau. Một controller gắn [Authorize(Policy = Policies.Instructor)] ở cấp class thì mọi action bên trong đều đòi vai trò giảng viên — kể cả action bạn vừa gắn thêm [Authorize(Policy = Policies.Student)], và kết quả là không ai vào được vì không tài khoản nào có cả hai vai. Endpoint mà sinh viên phải gọi được (xem bài nộp của mình, tự tham gia nhóm) không được nằm trong một class mang policy chỉ dành cho giảng viên: để class trống policy — SetFallbackPolicy đã bắt buộc đăng nhập — rồi gắn policy xuống từng action. Chồng policy chỉ an toàn khi policy hẹp nằm trọn trong policy rộng — hiện dự án không có cặp nào như vậy, nên quy tắc thực tế đơn giản: chỉ đặt policy ở cấp action.

namespace TeachHub.Domain.Groups;

public static class GroupAllocator
{
    /// <summary>
    /// Chia đều nhất có thể. Phần dư được rải sang các nhóm sẵn có thay vì gom
    /// thành một nhóm lẻ loi — nhóm 1 người không phải là bài tập nhóm.
    /// seed cố định để "xem trước" và "xác nhận" cho cùng kết quả.
    /// </summary>
    public static List<List<Guid>> Allocate(IReadOnlyList<Guid> enrollmentIds, int groupSize, int seed)
    {
        if (enrollmentIds.Count == 0) return [];
        if (groupSize < 1) throw new ArgumentOutOfRangeException(nameof(groupSize));

        var xao = enrollmentIds.OrderBy(_ => 0).ToList();
        var rng = new Random(seed);
        for (var i = xao.Count - 1; i > 0; i--)          // Fisher–Yates
        {
            var j = rng.Next(i + 1);
            (xao[i], xao[j]) = (xao[j], xao[i]);
        }

        var soNhom = Math.Max(1, (int)Math.Round(xao.Count / (double)groupSize, MidpointRounding.ToZero));
        var nhom = Enumerable.Range(0, soNhom).Select(_ => new List<Guid>()).ToList();

        // Chia vòng tròn: chênh lệch giữa nhóm lớn nhất và nhỏ nhất luôn ≤ 1.
        for (var i = 0; i < xao.Count; i++) nhom[i % soNhom].Add(xao[i]);

        return nhom;
    }
}

public sealed class GroupsController(ISender sender) : ApiControllerBase
{
    [HttpGet("offerings/{offeringId:guid}/groups")]
    [Authorize(Policy = Policies.Instructor)]
    [ProducesResponseType<IReadOnlyList<GroupDto>>(StatusCodes.Status200OK)]
    public async Task<IActionResult> GetByOffering(Guid offeringId, CancellationToken ct)
        => Ok(await sender.Send(new GetGroupsQuery(offeringId, CurrentUserId), ct));

    [HttpPost("offerings/{offeringId:guid}/groups")]
    [Authorize(Policy = Policies.Instructor)]
    public async Task<IActionResult> Create(
        Guid offeringId, CreateGroupCommand command, CancellationToken ct)
        => ToActionResult(await sender.Send(
            command with { OfferingId = offeringId, ActorId = CurrentUserId }, ct));

    [HttpPost("offerings/{offeringId:guid}/groups/auto-generate")]
    [Authorize(Policy = Policies.Instructor)]
    public async Task<IActionResult> AutoGenerate(
        Guid offeringId, AutoGenerateGroupsCommand command, CancellationToken ct)
        => ToActionResult(await sender.Send(
            command with { OfferingId = offeringId, ActorId = CurrentUserId }, ct));

    /// <summary>Sinh viên tự tham gia, nếu lớp cho phép.</summary>
    [HttpPost("groups/{id:guid}/join")]
    [Authorize(Policy = Policies.Student)]
    public async Task<IActionResult> Join(Guid id, CancellationToken ct)

    [HttpPut("groups/{id:guid}/members")]
    [Authorize(Policy = Policies.Instructor)]
    public async Task<IActionResult> UpdateMembers(

/// <summary>
/// ICommand chứ không phải IQuery: TransactionBehavior nhận diện qua marker IBaseCommand

public sealed record GradeSubmissionCommand(
    IReadOnlyDictionary<Guid, decimal> RubricScores,
    string? Feedback) : ICommand<Result<GradedSubmissionDto>>
{
    public Guid SubmissionId { get; init; }   // controller điền từ route

public sealed class GradeSubmissionCommandValidator : AbstractValidator<GradeSubmissionCommand>
{
    public GradeSubmissionCommandValidator()

        RuleFor(x => x.RubricScores).NotEmpty()

            .Must(kv => kv.Value >= 0m)
