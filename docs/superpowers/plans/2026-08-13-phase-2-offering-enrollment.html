public sealed class CreateOfferingCommandHandler(IAppDbContext db)
    : ICommandHandler<CreateOfferingCommand, Result<OfferingDto>>
{
    public async Task<Result<OfferingDto>> Handle(CreateOfferingCommand request, CancellationToken ct)
    {
        var versionId = request.SubjectVersionId
            ?? await db.Subjects.Where(s => s.Id == request.SubjectId)
                                .Select(s => s.CurrentVersionId)
                                .FirstOrDefaultAsync(ct);

        var status = await db.SubjectVersions
            .Where(v => v.Id == versionId)
            .Select(v => (SubjectVersionStatus?)v.Status)
            .FirstOrDefaultAsync(ct);

        if (status is null)
            return Result<OfferingDto>.NotFound("Không tìm thấy phiên bản môn học.");

        if (status == SubjectVersionStatus.Draft)
            return Result<OfferingDto>.Invalid("Không mở được lớp từ phiên bản đang ở trạng thái nháp.");

        // ... tạo CourseOffering, sinh join_code, SaveChangesAsync ...
    }

using System.Globalization;
using System.Text.RegularExpressions;

namespace TeachHub.Application.Features.Enrollments;

public sealed record ImportRow(string StudentCode, string FullName, string Email, string? ClassName);
public sealed record ImportError(int RowNumber, string Message);
public sealed record ValidatedStudent(string StudentCode, string FullName, string Email, string? ClassName);
public sealed record ImportResult(List<ValidatedStudent> Valid, List<ImportError> Errors);

public static partial class EnrollmentImporter
{
    /// <summary>
    /// Nhập được bao nhiêu nhập bấy nhiêu. Một dòng hỏng không được làm hỏng cả file —
    /// giảng viên chỉ cần sửa đúng những dòng được liệt kê rồi nhập bổ sung.
    /// </summary>
    public static ImportResult Validate(IReadOnlyList<ImportRow> rows)
    {
        var valid = new List<ValidatedStudent>();
        var errors = new List<ImportError>();
        var daThay = new HashSet<string>(StringComparer.OrdinalIgnoreCase);

        for (var i = 0; i < rows.Count; i++)
        {
            var soDong = i + 2;   // +1 vì 1-based, +1 vì dòng header
            var r = rows[i];

            var ma    = NormalizeStudentCode(r.StudentCode);
            var ten   = NormalizeName(r.FullName);
            var email = r.Email?.Trim().ToLowerInvariant() ?? "";

            if (ma.Length == 0 && ten.Length == 0 && email.Length == 0)
                continue;                                    // dòng trống: bỏ qua lặng lẽ

            if (ma.Length == 0)  { errors.Add(new(soDong, "Thiếu MSSV.")); continue; }
            if (ten.Length == 0) { errors.Add(new(soDong, $"Thiếu họ tên (MSSV {ma}).")); continue; }

            if (email.Length > 0 && !EmailPattern().IsMatch(email))
            { errors.Add(new(soDong, $"Email không hợp lệ: '{r.Email}'.")); continue; }

            if (!daThay.Add(ma))
            { errors.Add(new(soDong, $"MSSV {ma} bị trùng với một dòng phía trên.")); continue; }

            valid.Add(new ValidatedStudent(ma, ten, email, r.ClassName?.Trim()));
        }

        return new ImportResult(valid, errors);
    }

    /// <summary>
    /// Excel hay biến MSSV thành số rồi xuất ra "2251012001.0" hoặc dạng khoa học.
    /// Khôi phục về chuỗi chữ số gốc.
    /// </summary>
    private static string NormalizeStudentCode(string? raw)
    {
        var s = raw?.Trim() ?? "";
        if (s.Length == 0) return "";

        if (decimal.TryParse(s, NumberStyles.Float, CultureInfo.InvariantCulture, out var d)
            && d == Math.Floor(d))
            return ((long)d).ToString(CultureInfo.InvariantCulture);

        return s;
    }

    private static string NormalizeName(string? raw) =>
        WhitespaceRuns().Replace(raw?.Trim() ?? "", " ");

    [GeneratedRegex(@"\s+")]              private static partial Regex WhitespaceRuns();
    [GeneratedRegex(@"^[^@\s]+@[^@\s]+\.[^@\s]+$")] private static partial Regex EmailPattern();
}

public enum LessonAccess { Locked, Unlocked }

public sealed record ComponentNode(Guid Id, bool IsRequired);

public sealed record Progress(HashSet<Guid> CompletedLessons, HashSet<Guid> CompletedComponents)
{
    public static Progress Empty => new([], []);
    public static Progress With(IEnumerable<Guid>? completedLessons = null,

/// <summary>
/// Hàm thuần, không I/O. Tách khỏi database để test được mọi tổ hợp mở khóa trong mili-giây —
/// đây là loại logic mà sai một ca biên là sinh viên bị kẹt không học tiếp được, và bạn
/// sẽ nhận tin nhắn lúc 11 giờ đêm.
/// </summary>
public static class UnlockEngine
{
    public static Dictionary<Guid, LessonAccess> Evaluate(ChapterNode chapter, Progress progress)
    {
        var result = new Dictionary<Guid, LessonAccess>();

        if (chapter.Rule == UnlockRule.Free)
        {
            foreach (var l in chapter.Lessons) result[l.Id] = LessonAccess.Unlocked;
            return result;
        }

        var moKhoa = true;
        foreach (var lesson in chapter.Lessons)
        {
            result[lesson.Id] = moKhoa ? LessonAccess.Unlocked : LessonAccess.Locked;

            // Chỉ bài được đánh dấu bắt buộc mới có quyền chặn bài kế tiếp.
            if (moKhoa && lesson.IsRequiredToUnlockNext && !IsComplete(lesson, progress))
                moKhoa = false;
        }

        return result;
    }

    public static bool IsComplete(LessonNode lesson, Progress progress) =>
        progress.CompletedLessons.Contains(lesson.Id)
        && lesson.Components
            .Where(c => c.IsRequired)
            .All(c => progress.CompletedComponents.Contains(c.Id));

    public static decimal CompletionPercent(ChapterNode chapter, Progress progress)
    {
        if (chapter.Lessons.Count == 0) return 0m;

        var xong = chapter.Lessons.Count(l => IsComplete(l, progress));
        return Math.Round(xong * 100m / chapter.Lessons.Count, 1);
    }
}

LibreOffice là phụ thuộc hệ thống nặng. Nếu không muốn cài: chấp nhận cho PPTX chỉ tải về, và khuyến nghị giảng viên xuất PDF trước khi upload. Đây là chỗ cắt giảm hợp lý nhất của Phase 2 — hãy quyết sớm thay vì để nó chặn tiến độ. 
private static readonly string[] AllowedVideoHosts =
    ["youtube.com", "www.youtube.com", "youtu.be", "player.vimeo.com", "vimeo.com"];

Cho nhúng iframe tùy ý nghĩa là bất kỳ ai sửa được nội dung bài học đều có thể nhúng trang lừa đảo vào trang học của sinh viên.

Nghiệm thu Phase 2

  1. Mở lớp DA24TIN01 — HK1 2026-2027 từ môn đã xuất bản ở Phase 1; xác nhận lớp ghim đúng version.
  2. Import file Excel 40 sinh viên có 2 dòng trùng MSSV và 1 dòng thiếu họ tên → hệ thống nhập 37 dòng và liệt kê đúng 3 dòng lỗi kèm số dòng.
  3. Sửa 3 dòng đó trong file rồi import lại → đủ 40, không ai bị trùng.
  4. Sinh viên đăng ký, nhập mã lớp, vào được lớp. Người không có trong danh sách nhập đúng mã vẫn bị từ chối.
  5. Đặt chương 1 ở chế độ after_previous_completed, bài 1 bắt buộc → sinh viên thấy bài 2 bị khóa kèm lời giải thích rõ ràng.
  6. Học xong bài 1 → bài 2 mở, % hoàn thành chương cập nhật.
  7. Mở trang học trên điện thoại → đọc được, không phải cuộn ngang.
  8. Xem trước PDF ngay trong trình duyệt, không phải tải về.
