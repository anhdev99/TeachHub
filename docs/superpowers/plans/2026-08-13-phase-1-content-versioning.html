using TeachHub.Domain.Common;

namespace TeachHub.Domain.Content;

public enum SubjectVersionStatus { Draft, Published, Archived }

public class SubjectVersion : Entity
{
    private SubjectVersion() { }   // cho EF Core

    public Guid SubjectId { get; private set; }
    public string VersionLabel { get; private set; } = null!;
    public SubjectVersionStatus Status { get; private set; }
    public Syllabus Syllabus { get; set; } = new();
    public Guid? ClonedFromVersionId { get; private set; }
    public string? Changelog { get; set; }
    public DateTimeOffset? PublishedAt { get; private set; }
    public DateTimeOffset CreatedAt { get; private set; }

    public List<Chapter> Chapters { get; private set; } = [];

    /// <summary>
    /// Chỉ bản draft mới sửa được. Đây là bảo đảm cốt lõi khiến các lớp học phần
    /// đã mở không bị đổi nội dung dưới chân (SRS mục 5.2).
    /// </summary>
    public bool CanEditContent => Status == SubjectVersionStatus.Draft;

    public static SubjectVersion CreateDraft(Guid subjectId, string label, DateTimeOffset now,
                                             Guid? clonedFrom = null)
    {
        if (string.IsNullOrWhiteSpace(label))
            throw new ArgumentException("Nhãn phiên bản không được để trống.", nameof(label));

        return new SubjectVersion
        {
            SubjectId = subjectId,
            VersionLabel = label.Trim(),
            Status = SubjectVersionStatus.Draft,
            ClonedFromVersionId = clonedFrom,
            CreatedAt = now,
        };
    }

    public void Publish(DateTimeOffset now)
    {
        if (Status != SubjectVersionStatus.Draft)
            throw new InvalidOperationException(
                $"Chỉ xuất bản được bản nháp. Phiên bản '{VersionLabel}' đang ở trạng thái {Status}.");

        Status = SubjectVersionStatus.Published;
        PublishedAt = now;
    }

    public void Archive(DateTimeOffset now)
    {
        if (Status != SubjectVersionStatus.Published)
            throw new InvalidOperationException(
                $"Chỉ lưu trữ được phiên bản đã xuất bản. '{VersionLabel}' đang ở {Status}.");

        Status = SubjectVersionStatus.Archived;
    }
}

using TeachHub.Domain.Common;

namespace TeachHub.Domain.Content;

public enum LessonContentType { Ppt, Pdf, VideoEmbed, Text, Interactive }

public class Lesson : Entity, ISoftDeletable
{
    public Guid ChapterId { get; set; }
    public int OrderIndex { get; set; }
    public required string Title { get; set; }
    public LessonContentType ContentType { get; set; }
    public string? FileUrl { get; set; }
    public string? TextContent { get; set; }
    public int DurationMinutes { get; set; }

    /// <summary>Giảng viên bật/tắt cho TỪNG bài — không phải cài đặt toàn cục (SRS mục 3.2).</summary>
    public bool IsRequiredToUnlockNext { get; set; }

    public List<LessonComponent> Components { get; private set; } = [];

    public bool IsDeleted { get; set; }
    public DateTimeOffset? DeletedAt { get; set; }

    /// <summary>Không có ràng buộc nào bắt Lesson phải có component (SRS mục 5.1).</summary>
    public bool IsValid() => !string.IsNullOrWhiteSpace(Title);

    public LessonComponent AddComponent(LessonComponentType type, Guid? refId, ComponentConfig config)
    {
        if (LessonComponent.RequiresRef(type) && refId is null)
            throw new ArgumentException($"Component loại {type} bắt buộc phải trỏ tới một bản ghi.", nameof(refId));
        if (!LessonComponent.RequiresRef(type) && refId is not null)
            throw new ArgumentException($"Component loại {type} mang nội dung nhúng, không được có ref_id.", nameof(refId));

        var c = new LessonComponent
        {
            LessonId = Id, Type = type, RefId = refId,
            Config = config, OrderIndex = Components.Count,
        };
        Components.Add(c);
        return c;
    }

    public void RemoveComponent(Guid componentId)
    {
        var c = Components.SingleOrDefault(x => x.Id == componentId)
                ?? throw new InvalidOperationException($"Không có component {componentId} trong bài này.");
        Components.Remove(c);
        Reindex();
    }

    public void MoveComponent(Guid componentId, int toIndex)
    {
        var c = Components.SingleOrDefault(x => x.Id == componentId)
                ?? throw new InvalidOperationException($"Không có component {componentId} trong bài này.");

        Components.Remove(c);
        Components.Insert(Math.Clamp(toIndex, 0, Components.Count), c);
        Reindex();
    }

    // OrderIndex phải luôn là 0..n-1 liên tục. Nếu để hở, kéo-thả ở UI sẽ
    // dần tạo ra các khoảng trống và thứ tự hiển thị thành không xác định.
    private void Reindex()
    {
        for (var i = 0; i < Components.Count; i++) Components[i].OrderIndex = i;
    }
}

/// <summary>
/// Ghi dữ liệu ⇒ khai là ICommand chứ không phải IQuery. Đây không phải hình thức:
/// TransactionBehavior chỉ mở transaction cho request cài IBaseCommand, nên khai sai
/// kiểu là mất luôn tính nguyên tử của bước đổi trạng thái bên dưới.
/// </summary>
public sealed record PublishVersionCommand(Guid VersionId, Guid InstructorId) : ICommand<Result>;

public sealed class PublishVersionCommandHandler(IAppDbContext db, TimeProvider clock)
    : ICommandHandler<PublishVersionCommand, Result>
{
    public async Task<Result> Handle(PublishVersionCommand request, CancellationToken ct)
    {
        var (versionId, instructorId) = request;

        var version = await db.SubjectVersions
            .SingleOrDefaultAsync(v => v.Id == versionId, ct);

        if (version is null) return Result.NotFound();

        var subject = await db.Subjects
            .SingleOrDefaultAsync(s => s.Id == version.SubjectId && s.InstructorId == instructorId, ct);

        // Không phải chủ sở hữu ⇒ trả 404 chứ không 403: 403 vô tình xác nhận
        // "id này có tồn tại", đủ để dò ra cấu trúc dữ liệu của giảng viên khác.
        if (subject is null) return Result.NotFound();

        if (version.Status != SubjectVersionStatus.Draft)
            return Result.Conflict($"Phiên bản đang ở trạng thái {version.Status}, chỉ xuất bản được bản nháp.");

        var now = clock.GetUtcNow();

        // Toàn bộ chuyển trạng thái nằm trong một SaveChanges, và TransactionBehavior
        // đã bọc sẵn cả handler trong một transaction. Nếu tách ra nhiều lần lưu,
        // một lỗi giữa chừng sẽ để lại hai bản published cùng lúc.
        var dangPublished = await db.SubjectVersions
            .Where(v => v.SubjectId == version.SubjectId && v.Status == SubjectVersionStatus.Published)
            .ToListAsync(ct);

        foreach (var cu in dangPublished) cu.Archive(now);

        version.Publish(now);
        subject.CurrentVersionId = version.Id;

        await db.SaveChangesAsync(ct);
        return Result.Ok();
    }
}

using Microsoft.AspNetCore.Mvc;
using Microsoft.AspNetCore.Mvc.Filters;

namespace TeachHub.Api.Filters;

/// <summary>
/// Chặn mọi thao tác sửa nội dung khi SubjectVersion không còn ở trạng thái draft.
/// Đặt trên action, hoặc trên cả controller nếu mọi action đều là thao tác sửa.
/// </summary>
[AttributeUsage(AttributeTargets.Class | AttributeTargets.Method)]
public sealed class RequireDraftVersionAttribute : Attribute, IFilterFactory
{
    public bool IsReusable => false;

    public IFilterMetadata CreateInstance(IServiceProvider services) =>
        ActivatorUtilities.CreateInstance<Filter>(services);

    private sealed class Filter(AppDbContext db) : IAsyncActionFilter
    {
        public async Task OnActionExecutionAsync(ActionExecutingContext ctx, ActionExecutionDelegate next)
        {
            var ct = ctx.HttpContext.RequestAborted;

            // Lần ngược từ bất kỳ id nào có trong route về SubjectVersion.
            var status = await ResolveVersionStatusAsync(db, ctx.RouteData.Values, ct);

            if (status is null)
            {
                ctx.Result = new NotFoundResult();
                return;
            }

            if (status != SubjectVersionStatus.Draft)
            {
                ctx.Result = new ConflictObjectResult(new ProblemDetails
                {
                    Status = StatusCodes.Status409Conflict,
                    Title = "Phiên bản không sửa được",
                    Detail = $"Phiên bản đang ở trạng thái {status}, chỉ bản nháp mới sửa được nội dung.",
                });
                return;
            }

            await next();
        }
    }
}

/// <summary>
/// Nhân bản toàn bộ nội dung của một SubjectVersion sang một bản draft mới (ADR-004).
///
/// Điểm mấu chốt: LessonComponent.RefId trỏ tới Assignment/Quiz. Nếu copy nguyên giá trị
/// cũ, bản clone sẽ dùng chung Assignment với bản gốc — sửa bài tập ở năm mới sẽ đổi luôn
/// bài tập của lớp năm cũ. Vì vậy phải clone Assignment/Quiz TRƯỚC, dựng bảng ánh xạ
/// id cũ → id mới, rồi mới remap RefId.
/// </summary>
public sealed class SubjectVersionCloner(IAppDbContext db, TimeProvider clock)
{
    public async Task<SubjectVersion> CloneAsync(Guid sourceVersionId, string newLabel, CancellationToken ct)
    {
        var now = clock.GetUtcNow();

        var source = await db.SubjectVersions
            .AsNoTracking()
            .SingleAsync(v => v.Id == sourceVersionId, ct);

        var chapters = await db.Chapters.AsNoTracking()
            .Where(c => c.SubjectVersionId == sourceVersionId)
            .OrderBy(c => c.OrderIndex).ToListAsync(ct);

        var chapterIds = chapters.Select(c => c.Id).ToList();
        var lessons = await db.Lessons.AsNoTracking()
            .Where(l => chapterIds.Contains(l.ChapterId))
            .OrderBy(l => l.OrderIndex).ToListAsync(ct);

        var lessonIds = lessons.Select(l => l.Id).ToList();
        var components = await db.LessonComponents.AsNoTracking()
            .Where(c => lessonIds.Contains(c.LessonId))
            .OrderBy(c => c.OrderIndex).ToListAsync(ct);

        var assignments = await db.Assignments.AsNoTracking()
            .Where(a => a.SubjectVersionId == sourceVersionId).ToListAsync(ct);
        var quizzes = await db.Quizzes.AsNoTracking()
            .Where(q => q.SubjectVersionId == sourceVersionId).ToListAsync(ct);

        var target = SubjectVersion.CreateDraft(source.SubjectId, newLabel, now, clonedFrom: source.Id);
        target.Syllabus = source.Syllabus.DeepCopy();
        db.SubjectVersions.Add(target);

        // id cũ → id mới, cho MỌI bản ghi được nhân bản
        var map = new Dictionary<Guid, Guid>();

        // 1) Assignment và Quiz phải clone TRƯỚC — component sẽ tham chiếu tới chúng.
        foreach (var a in assignments)
        {
            var copy = a.CloneTo(target.Id);
            map[a.Id] = copy.Id;
            db.Assignments.Add(copy);
        }
        foreach (var q in quizzes)
        {
            var copy = q.CloneTo(target.Id);
            map[q.Id] = copy.Id;
            db.Quizzes.Add(copy);
        }

        // 2) Chương → bài → component, remap FK theo bảng ánh xạ.
        foreach (var ch in chapters)
        {
            var chCopy = new Chapter
            {
                SubjectVersionId = target.Id,
                Title = ch.Title, OrderIndex = ch.OrderIndex,
                Objective = ch.Objective,
                FoundationTags = [.. ch.FoundationTags],
                UnlockRule = ch.UnlockRule,
            };
            map[ch.Id] = chCopy.Id;
            db.Chapters.Add(chCopy);

            foreach (var ls in lessons.Where(l => l.ChapterId == ch.Id))
            {
                var lsCopy = new Lesson
                {
                    ChapterId = chCopy.Id,
                    Title = ls.Title, OrderIndex = ls.OrderIndex,
                    ContentType = ls.ContentType,
                    FileUrl = ls.FileUrl,          // dùng chung file đã upload — cố ý,
                    TextContent = ls.TextContent,  // nhân bản file PPT/PDF là lãng phí vô nghĩa
                    DurationMinutes = ls.DurationMinutes,
                    IsRequiredToUnlockNext = ls.IsRequiredToUnlockNext,
                };
                map[ls.Id] = lsCopy.Id;
                db.Lessons.Add(lsCopy);

                foreach (var cp in components.Where(c => c.LessonId == ls.Id))
                {
                    db.LessonComponents.Add(new LessonComponent
                    {
                        LessonId = lsCopy.Id,
                        Type = cp.Type,
                        OrderIndex = cp.OrderIndex,
                        IsRequired = cp.IsRequired,
                        Config = cp.Config.DeepCopy(),

                        // ★ Dòng quan trọng nhất của cả phase.
                        RefId = cp.RefId is null ? null
                              : map.TryGetValue(cp.RefId.Value, out var moi) ? moi
                              : throw new InvalidOperationException(
                                  $"Component {cp.Id} trỏ tới {cp.RefId} nhưng bản ghi đó không thuộc " +
                                  $"version nguồn {sourceVersionId}. Dữ liệu không nhất quán, dừng clone."),
                    });
                }
            }
        }

        await db.SaveChangesAsync(ct);
        return target;
    }
}

/// <summary>
/// Kiểm tra chữ ký byte đầu file thay vì tin phần mở rộng hay Content-Type do client gửi —
/// đúng yêu cầu NFR "validate loại file thực tế" (SRS mục 9). Cùng lớp này sẽ được
/// Phase 3 dùng lại cho file sinh viên nộp.
/// </summary>
public static class FileSignatureValidator
{
    private static readonly Dictionary<string, byte[][]> Signatures = new(StringComparer.OrdinalIgnoreCase)
    {
        [".pdf"]  = [[0x25, 0x50, 0x44, 0x46]],                       // %PDF
        [".zip"]  = [[0x50, 0x4B, 0x03, 0x04], [0x50, 0x4B, 0x05, 0x06]],
        [".pptx"] = [[0x50, 0x4B, 0x03, 0x04]],                       // OOXML là zip
        [".docx"] = [[0x50, 0x4B, 0x03, 0x04]],
        [".png"]  = [[0x89, 0x50, 0x4E, 0x47]],
        [".jpg"]  = [[0xFF, 0xD8, 0xFF]],
        [".jpeg"] = [[0xFF, 0xD8, 0xFF]],
        [".mp4"]  = [[0x00, 0x00, 0x00]],                             // ftyp nằm ở offset 4
    };

    public static bool Matches(string fileName, ReadOnlySpan<byte> head)
    {
        var ext = Path.GetExtension(fileName);
        if (!Signatures.TryGetValue(ext, out var sigs)) return false;

        foreach (var sig in sigs)
            if (head.Length >= sig.Length && head[..sig.Length].SequenceEqual(sig)) return true;

        return false;
    }
}
